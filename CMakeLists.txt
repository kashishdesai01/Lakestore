cmake_minimum_required(VERSION 3.24)
project(lakestore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(FetchContent)
option(LAKESTORE_CLOUD "Build S3 and Azure REST adapters" OFF)
set(LAKESTORE_SANITIZER "" CACHE STRING "address, undefined, address,undefined or thread")
if(LAKESTORE_SANITIZER)
  add_compile_options(-fsanitize=${LAKESTORE_SANITIZER} -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=${LAKESTORE_SANITIZER})
endif()
find_package(Threads REQUIRED)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
FetchContent_MakeAvailable(json)
add_library(lakestore_lib src/objstore/store.cpp src/format/format.cpp src/log/log.cpp src/table/table.cpp)
target_include_directories(lakestore_lib PUBLIC include)
target_link_libraries(lakestore_lib PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(lakestore_lib PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(LAKESTORE_CLOUD)
  find_package(CURL 7.75 REQUIRED)
  find_package(OpenSSL REQUIRED)
  find_package(LibXml2 REQUIRED)
  target_sources(lakestore_lib PRIVATE src/objstore/cloud.cpp)
  target_link_libraries(lakestore_lib PRIVATE CURL::libcurl OpenSSL::Crypto LibXml2::LibXml2)
  target_compile_definitions(lakestore_lib PUBLIC LAKESTORE_CLOUD=1)
endif()
add_executable(lakestore src/cli/main.cpp)
target_link_libraries(lakestore PRIVATE lakestore_lib)
target_compile_options(lakestore PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz URL_HASH SHA256=7b42b4d6ed48810c5362c265a17faebe90dc2373c885e5216439d37927f02926)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)
  add_executable(lakestore_tests tests/store_test.cpp tests/format_test.cpp tests/table_test.cpp tests/model_test.cpp tests/stream_test.cpp)
  target_link_libraries(lakestore_tests PRIVATE lakestore_lib GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(lakestore_tests DISCOVERY_TIMEOUT 30)
  if(LAKESTORE_CLOUD)
    add_executable(lakestore_cleanup tests/cloud_cleanup.cpp)
    target_link_libraries(lakestore_cleanup PRIVATE lakestore_lib)
    add_executable(lakestore_cloud_tests tests/cloud_test.cpp)
    target_link_libraries(lakestore_cloud_tests PRIVATE lakestore_lib GTest::gtest_main)
    add_test(NAME cloud COMMAND lakestore_cloud_tests)
    set_tests_properties(cloud PROPERTIES LABELS "cloud" SKIP_RETURN_CODE 77)
  endif()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME cli_smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_smoke.py $<TARGET_FILE:lakestore>)
endif()
install(TARGETS lakestore lakestore_lib)
install(DIRECTORY include/ DESTINATION include)
